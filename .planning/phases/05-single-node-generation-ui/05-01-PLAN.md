          criticScore: state.criticScore,
          isReplaceable: true,
          replacementTags: [],
          actVariant: false,
        });

        // Navigate to node detail
        navigate(`/nodes/${savedNode.id}`);

        // Reset form and state
        reset();
        form.reset();
      } catch (error) {
        toast.error(error instanceof Error ? error.message : 'Failed to save node');
      }
    };
