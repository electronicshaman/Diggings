    ```typescript
      if (!state.content) return;

      try {
        // Build node data from form values + generated content
        const formValues = form.getValues();
        const savedNode = await createNodeMutation.mutateAsync({
          type: formValues.nodeType,
          biome: formValues.biome,
          name: formValues.name,
          acts: formValues.acts as [number, ...number[]],
          themes: biomeConfig?.themes?.slice(0, 2) || ['survival'],
          entityTypes: ['character'],
          content: state.content,
          criticScore: state.criticScore,
          isReplaceable: true,
          replacementTags: [],
          actVariant: false,
        });

        toast({
          title: 'Node saved!',
          description: `${savedNode.name} has been created.`,
        });

        // Navigate to node detail
        navigate(`/nodes/${savedNode.id}`);

        // Reset form and state
        reset();
        form.reset();
      } catch (error) {
        toast({
          variant: 'destructive',
          title: 'Save failed',
          description: error instanceof Error ? error.message : 'Unknown error',
        });
      }
    };
    ```
