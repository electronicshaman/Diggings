Add a new function `generateBeatList` that:

2. Returns Promise<{ beats: Array<{ id: string, role: string, text: string }> }>
3. Is NOT streaming (returns JSON directly)
4. Handles errors with proper error state

```typescript
const generateBeatList = useCallback(async (context: FieldContext): Promise<GeneratedBeat[] | null> => {
  setError(null);

  try {
    const response = await fetch('/api/generate/field/beat-list', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        nodeType: context.nodeType,
        biome: context.biome,
        name: context.name,
        themes: context.themes || [],
        narrativeHook: context.narrativeHook,
      }),
    });

    if (!response.ok) {
      const error = await response.json();
      throw new Error(error.error || `HTTP ${response.status}`);
    }

    const data = await response.json();
    return data.beats;
  } catch (err) {
    const message = err instanceof Error ? err.message : 'Failed to generate beats';
    setError(message);
    options?.onError?.(message);
    return null;
  }
}, [options]);
```

```typescript
const handleSuggestBeats = async () => {
  const context = {
    nodeType: formData.type,
    biome: formData.biome,
    name: formData.name,
    themes: formData.themes,
    narrativeHook: formData.content?.narrative_hook,
  };

  if (!context.nodeType || !context.biome) {
    toast.warning('Cannot suggest beats', {
      description: 'Please complete node type and biome first',
    });
    return;
  }

  setIsGeneratingBeats(true);
  const beats = await generateBeatList(context);
  setIsGeneratingBeats(false);

  if (beats) {
    // Append to existing beats with unique IDs
    const currentBeats = formData.content?.beats || [];
    const newBeats = beats.map((b, i) => ({
      ...b,
      id: `beat_${Date.now()}_${i}`,
    }));

    // Update form data
    setFormData({
      ...formData,
      content: {
        ...formData.content,
        beats: [...currentBeats, ...newBeats],
      },
    });

    toast.success('Beats suggested', {
      description: `Added ${beats.length} story beats. Edit as needed.`,
    });
  }
};
```
